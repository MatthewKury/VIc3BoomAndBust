# ================================================================
#  Boom & Bust — Journal Entry Scripted Buttons
#
#  bb_dissolve_monopoly_button (attached to je_bb_financial)
#    Visible when the Monopoly Power bar reads >= 40 — checked
#    directly on the bar, so there is no monthly-pulse lag.
#    Greyed out (possible) while a dissolution case is running
#    or no interest group is in government.
#    Costs decaying authority over 12 months, then opens bb.10
#    for IG-leader selection.
#    ai_chance base 0: the JE is active for every country, so the
#    AI must be explicitly barred from pressing the button.
# ================================================================

# ── Adjust Fiscal Temperature ───────────────────────────────────
# Always visible on the JE. Opens bb.20, where the player picks one
# of four stances that push the Risk Stance bar up or down monthly
# for 5 years (stored in the timed variable bb_fiscal_temp, read by
# bb_risk_stance_bar). The button stays greyed out while a stance
# is in force; bb.21 notifies the player when it lapses.

bb_fiscal_temperature_button = {
    name = "bb_fiscal_temperature_button"
    desc = "bb_fiscal_temperature_button_desc"

    visible = { always = yes }

    possible = {
        NOT = { has_variable = bb_fiscal_temp }
    }

    ai_chance = {
        value = 0
    }

    effect = {
        trigger_event = { id = bb.20 popup = yes }
    }
}

bb_dissolve_monopoly_button = {
    name = "bb_dissolve_monopoly_button"
    desc = "bb_dissolve_monopoly_button_desc"

    visible = {
        scope:journal_entry = {
            "scripted_bar_progress(bb_monopoly_power_bar)" >= 40
        }
    }

    possible = {
        NOT = { has_variable = bb_case_strength }
        any_interest_group = {
            is_in_government = yes
        }
    }

    ai_chance = {
        value = 0
    }

    effect = {
        add_modifier = {
            name = bb_antitrust_case_cost
            months = 12
            is_decaying = yes
        }
        trigger_event = { id = bb.10 popup = yes }
    }
}

# ── Appoint a Bank Governor ─────────────────────────────────────
# Visible while the seat is empty (the bb_bank_governor trait is
# the office record — see bb_traits.txt). Opens bb.40, which
# offers candidates drawn live from the character pool.

bb_appoint_governor_button = {
    name = "bb_appoint_governor_button"
    desc = "bb_appoint_governor_button_desc"

    visible = {
        NOT = {
            any_scope_character = { has_trait = bb_bank_governor }
        }
    }

    ai_chance = {
        value = 0
    }

    effect = {
        trigger_event = { id = bb.40 popup = yes }
    }
}

# ── Dismiss the Governor ────────────────────────────────────────
# Mutually exclusive with the appoint button. Sacking the governor
# angers their interest group and drains authority for a year —
# unless the government was recently reformed or re-elected
# (bb_fresh_mandate, set by bb.3 for 12 months): a new government
# naming its own banker is ordinary housekeeping, so the change
# costs only a brief administrative backlash and no radicals.

bb_dismiss_governor_button = {
    name = "bb_dismiss_governor_button"
    desc = "bb_dismiss_governor_button_desc"

    visible = {
        any_scope_character = { has_trait = bb_bank_governor }
    }

    ai_chance = {
        value = 0
    }

    effect = {
        random_scope_character = {
            limit = { has_trait = bb_bank_governor }
            save_scope_as = bb_gov_out
        }
        if = {
            limit = { has_variable = bb_fresh_mandate }
            add_modifier = {
                name = bb_governor_dismissal_backlash
                months = 3
                is_decaying = yes
            }
        }
        else = {
            add_radicals = {
                value = small_radicals
                interest_group = scope:bb_gov_out.interest_group
            }
            add_modifier = {
                name = bb_governor_dismissal_backlash
                months = 12
                is_decaying = yes
            }
        }
        scope:bb_gov_out = {
            remove_trait = bb_bank_governor
        }
        bb_clear_governor_office = yes
    }
}

# ── National Bank Intervention (lender of last resort) ──────────
# Visible only while Market Health < 40. Opens bb.50: pump treasury
# money into the markets at the price of moral hazard (a lasting
# upward pull on Risk Stance). 24-month cooldown keeps it a crisis
# tool rather than a faucet.

bb_intervene_button = {
    name = "bb_intervene_button"
    desc = "bb_intervene_button_desc"

    visible = {
        scope:journal_entry = {
            "scripted_bar_progress(bb_market_health_bar)" < 40
        }
    }

    cooldown = { months = 24 }

    ai_chance = {
        value = 0
    }

    effect = {
        trigger_event = { id = bb.50 popup = yes }
    }
}

# ── Suspend the Exchange (market holiday) ───────────────────────
# Panic-band emergency brake, visible only while Market Health
# < 20. For 4 months bb_roll_market_event skips all shocks and
# major-event rolls (the underlying trend keeps running — closing
# the doors buys time, it does not fix fundamentals). Capitalists
# are furious and the closure itself drags on the economy.

bb_market_holiday_button = {
    name = "bb_market_holiday_button"
    desc = "bb_market_holiday_button_desc"

    visible = {
        scope:journal_entry = {
            "scripted_bar_progress(bb_market_health_bar)" < 20
        }
    }

    cooldown = { months = 36 }

    ai_chance = {
        value = 0
    }

    effect = {
        set_variable = { name = bb_market_holiday months = 4 }
        set_variable = { name = bb_health_shock value = 0 }
        add_modifier = {
            name = bb_exchange_closed
            months = 4
        }
        add_radicals = {
            value = medium_radicals
            pop_type = capitalists
        }
        add_radicals = {
            value = small_radicals
            pop_type = shopkeepers
        }
        post_notification = bb_news_market_holiday
    }
}
