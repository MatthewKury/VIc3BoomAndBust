# ================================================================
#  Boom & Bust — Decisions
#
#  bb_dissolve_monopoly_decision
#    Visible when the Monopoly Power bar is in Present/Influential/
#    Dominant territory (bar >= 40, detected via active modifier).
#    Costs decaying authority over 12 months.
#    Opens bb.10 for IG-leader selection.
#    Cooldown: cannot be re-used while a dissolution is still active.
# ================================================================

bb_dissolve_monopoly_decision = {
    is_shown = {
        is_local_player = yes
        has_journal_entry = je_bb_financial
        OR = {
            has_modifier = bb_monopoly_present
            has_modifier = bb_monopoly_influential
            has_modifier = bb_monopoly_dominant
        }
    }

    possible = {
        NOT = {
            OR = {
                has_modifier = bb_monopoly_dissolution
                has_modifier = bb_monopoly_dissolution_strong
            }
        }
        any_interest_group = {
            is_in_government = yes
        }
    }

    when_taken = {
        add_modifier = {
            name = bb_antitrust_case_cost
            months = 12
            is_decaying = yes
        }
        trigger_event = { id = bb.10 }
    }

    ai_chance = { value = 0 }
}
