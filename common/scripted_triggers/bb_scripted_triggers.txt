# ================================================================
#  Boom & Bust — Scripted Triggers
#
#  Character-scope trait classifiers for the Risk Stance bar.
#  A character with traits from both sets contributes to both
#  directions and nets out to zero — intentional.
# ================================================================

# Temperaments that embolden investors (push Risk Stance up)
bb_char_heats_risk = {
    OR = {
        has_trait = reckless
        has_trait = ambitious
        has_trait = romantic
        has_trait = hedonist
        has_trait = grifter
    }
}

# Temperaments that counsel restraint (push Risk Stance down)
bb_char_cools_risk = {
    OR = {
        has_trait = cautious
        has_trait = meticulous
        has_trait = reserved
        has_trait = pious
        has_trait = erudite
    }
}

# ── Bank Governor trait classes ─────────────────────────────────

# Vices that expose a sitting Bank Governor to the bb.43 scandal
# roll in bb_governor_admin_pulse.
bb_char_gov_vice = {
    OR = {
        has_trait = grifter
        has_trait = expensive_tastes
        has_trait = alcoholic
        has_trait = opium_addiction
        has_trait = cocaine_addiction
    }
}

# Communicators set the national mood: a governor with one of
# these amplifies their temperament row on the Risk Stance bar
# (0.35 instead of 0.2) and can jawbone the markets in bb.50.
bb_char_gov_communicator = {
    OR = {
        has_trait = charismatic
        has_trait = inspirational_orator
        has_trait = demagogue
    }
}
