# -- Letters of Patent -----------------------------------------
# Starting option. The government grants exclusive charters,
# accelerating monopoly formation by 50%.
# Monopolists supply military goods at preferential rates (-10%)
# and subsidise construction inputs (-5%).
# Cannot be enacted under laws that oppose state-backed monopolies.

law_bb_letters_of_patent = {
    group = lawgroup_bb_antitrust
    icon  = "gfx/interface/icons/law_icons/interventionism.dds"
    progressiveness = 0

    disallowing_laws = {
        law_laissez_faire
        law_cooperative_ownership
        law_command_economy

    on_activate = {}

    modifier = {
        country_military_goods_cost_mult      = -0.10
        country_construction_goods_cost_mult  = -0.05
    }

    ai_enact_weight_modifier = { value = 0 }
}
