# ============================================================
#  Boom & Bust — Main Journal Entry
#
#  possible = { hidden_trigger { always = yes } } auto-activates
#  this JE for all countries on the first monthly pulse. Effects
#  in immediate and on_monthly_pulse are gated with is_local_player
#  so only the human player receives modifiers.
#
#  Hosts three scripted progress bars (defined in
#  common/scripted_progress_bars/bb_progress_bars.txt).
# ============================================================

je_bb_financial = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"

    group = je_group_internal_affairs

    weight = 100

    should_be_pinned_by_default_involved = yes
    should_be_pinned_by_default_uninvolved_or_context = yes

    is_shown_when_inactive = {
        is_player = yes
    }

    scripted_progress_bar = bb_market_health_bar
    scripted_progress_bar = bb_risk_stance_bar
    scripted_progress_bar = bb_monopoly_power_bar

    scripted_button = bb_fiscal_temperature_button
    scripted_button = bb_dissolve_monopoly_button
    scripted_button = bb_appoint_governor_button
    scripted_button = bb_dismiss_governor_button
    scripted_button = bb_intervene_button
    scripted_button = bb_market_holiday_button

    possible = {
        hidden_trigger = { always = yes }
    }

    complete = {
        hidden_trigger = { always = no }
    }

    # Apply starting modifiers only for the human player, and
    # announce the Financial Monitor with the opening-trend
    # choice (bb.1).
    immediate = {
        if = {
            limit = { is_local_player = yes }
            add_modifier = { name = bb_market_stable days = -1 }
            add_modifier = { name = bb_stance_tolerant days = -1 }
            bb_update_market_relations_cap = yes
            trigger_event = { id = bb.1 days = 3 }
        }
    }

    # Roll the month's market event, then re-evaluate and
    # re-apply band modifiers — player only
    on_monthly_pulse = {
        effect = {
            if = {
                limit = { is_local_player = yes }
                bb_update_market_relations_cap = yes
                bb_roll_market_event = yes
                bb_governor_admin_pulse = yes
                bb_apply_market_health_modifier = yes
                bb_apply_risk_stance_modifier = yes
                bb_apply_monopoly_modifier = yes
            }
        }
    }
}
