# ── bb.1: The Age of Capital (JE launch + inaugural governor) ─
# Fired from je_bb_financial's immediate block (player only,
# days = 3). Introduces the journal entry's theme, then puts the
# government's first duty on the table: naming the inaugural
# Governor of the National Bank. Same candidate machinery as
# bb.40 (bb_find_governor_candidates / bb_seat_governor); the
# old opening-trend choice was removed 2026-07-12 — the market
# now opens on whatever hand takes the tiller.

bb.1 = {
    type = country_event
    placement = ROOT
    title = bb.1.t

    duration = 3

    event_image = {
        video = "europenorthamerica_capitalists_meeting"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    immediate = {
        bb_find_governor_candidates = yes
    }

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_law_or_variant = law_type:law_hereditary_bureaucrats }
                desc = bb.1.desc.hereditary
            }
            triggered_desc = {
                trigger = { has_law = law_type:law_elected_bureaucrats }
                desc = bb.1.desc.elected
            }
            triggered_desc = {
                trigger = { always = yes }
                desc = bb.1.desc.appointed
            }
        }
    }

    # The financier (Appointed Bureaucrats)
    option = {
        name = bb.1.a
        trigger = { exists = scope:bb_gov_cand_fin }
        custom_tooltip = bb.1.a.tt
        bb_seat_governor = { CANDIDATE = bb_gov_cand_fin }
    }

    # The statesman (Appointed Bureaucrats)
    option = {
        name = bb.1.b
        trigger = { exists = scope:bb_gov_cand_gov }
        custom_tooltip = bb.1.b.tt
        bb_seat_governor = { CANDIDATE = bb_gov_cand_gov }
    }

    # The scholar (Appointed Bureaucrats)
    option = {
        name = bb.1.c
        trigger = { exists = scope:bb_gov_cand_sch }
        custom_tooltip = bb.1.c.tt
        bb_seat_governor = { CANDIDATE = bb_gov_cand_sch }
    }

    # The patrician (Hereditary Bureaucrats / Crownland Diets)
    option = {
        name = bb.1.d
        trigger = { exists = scope:bb_gov_cand_heir }
        custom_tooltip = bb.1.d.tt
        bb_seat_governor = { CANDIDATE = bb_gov_cand_heir }
    }

    # The administrator (Elected Bureaucrats)
    option = {
        name = bb.1.e
        trigger = { exists = scope:bb_gov_cand_civil }
        custom_tooltip = bb.1.e.tt
        bb_seat_governor = { CANDIDATE = bb_gov_cand_civil }
    }

    # Leave the seat empty for now
    option = {
        name = bb.1.z
        default_option = yes

# ── bb.10: Dissolve Monopoly — Governor Confirmation ─────────
# Triggered by bb_dissolve_monopoly_button, which requires a
# seated Bank Governor. The Governor is always the champion —
# no player choice of IG anymore. scope:bb_case_leader/bb_case_ig
# are set in immediate from the sitting governor and their IG,
# then read by bb_apply_dissolution_effect. Flavor text varies
# by the Governor's actual IG affiliation.

bb.10 = {
    type = country_event
    placement = ROOT
    title = bb.10.t

    duration = 3

    event_image = {
        video = "europenorthamerica_capitalists_meeting"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    immediate = {
        random_scope_character = {
            limit = { has_trait = bb_bank_governor }
            save_scope_as = bb_case_leader
        }
        scope:bb_case_leader = {
            interest_group = { save_scope_as = bb_case_ig }
        }
    }

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:bb_case_ig = { is_interest_group_type = ig_industrialists }
                }
                desc = bb.10.desc.industrialists
            }
            triggered_desc = {
                trigger = {
                    scope:bb_case_ig = { is_interest_group_type = ig_landowners }
                }
                desc = bb.10.desc.landowners
            }
            triggered_desc = {
                trigger = {
                    scope:bb_case_ig = { is_interest_group_type = ig_trade_unions }
                }
                desc = bb.10.desc.unions
            }
            triggered_desc = {
                trigger = { always = yes }
                desc = bb.10.desc.default
            }
        }
    }

    option = {
        name = bb.10.a
        bb_apply_dissolution_effect = yes
        trigger_event = { id = bb.11 days = 14 }
    }
}

# ── bb.20: Adjust Fiscal Temperature — Stance Selection ─────
# Triggered by bb_fiscal_temperature_button.
# Each option sets bb_fiscal_temp (monthly Risk Stance bar delta)
# for 60 months and schedules the bb.21 expiry notice in 5 years.
# The button is greyed out while bb_fiscal_temp exists.

bb.20 = {
    type = country_event
    placement = ROOT
    title = bb.20.t
    desc  = bb.20.desc

    duration = 3

    event_image = {
        video = "unspecific_politicians_arguing"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    # Strong heating: flood the market with cheap credit
    option = {
        name = bb.20.a
        custom_tooltip = bb.20.a.tt
        set_variable = {
            name = bb_fiscal_temp
            value = 1
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Mild heating: encourage speculative lending
    option = {
        name = bb.20.b
        custom_tooltip = bb.20.b.tt
        set_variable = {
            name = bb_fiscal_temp
            value = 0.5
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Mild cooling: tighten lending standards
    option = {
        name = bb.20.c
        custom_tooltip = bb.20.c.tt
        set_variable = {
            name = bb_fiscal_temp
            value = -0.5
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Strong cooling: impose strict capital discipline
    option = {
        name = bb.20.d
        default_option = yes
        custom_tooltip = bb.20.d.tt
        set_variable = {
            name = bb_fiscal_temp
            value = -1
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }
}

# ── bb.41: Governor Died in Office (hidden, on_character_death) ─

# Character — the one who just died. Empties the seat the instant
# its occupant dies, instead of waiting on bb_governor_admin_pulse's
# next monthly vacancy check (which stays in place as a fallback
# for other disappearance causes, e.g. exile). Hidden events must
# not have options.

bb.41 = {
    type = character_event
    hidden = yes

    trigger = {
        has_trait = bb_bank_governor

    }

    immediate = {
        remove_trait = bb_bank_governor
        owner = {
            bb_clear_governor_office = yes
            bb_market_jolt = { VALUE = -2 }
            trigger_event = { id = bb.42 popup = yes }
        }
    }
}

# ── bb.60: Open-Market Operations ───────────────────────────
# Opened by the Open-Market Operations button. The National Bank
# leans on the market in one direction or the other by planting a
# fresh bb_market_trend (the same drift variable market events
# use) plus an immediate jolt. Supporting a weak market costs
# money and breeds a little moral hazard; cooling an overheated
# one costs money and political comfort but adds no hazard — it is
# the deliberate counter to a runaway boom before the Risk Stance
# multiplier turns a mania into a crash.

bb.60 = {
    type = country_event
    placement = ROOT
    title = bb.60.t
    desc  = bb.60.desc

    duration = 3

    event_image = {
        video = "europenorthamerica_capitalists_meeting"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    # Support the market: open-market purchases. The cost is a
    # standing weekly outlay + authority for the run of the
    # operation (bb_market_support_ops, 9 months), not a lump sum.
    option = {
        name = bb.60.a
        custom_tooltip = bb.60.a.tt
        add_modifier = {
            name = bb_market_support_ops
            months = 9
        }
        bb_market_jolt = { VALUE = 2 }
        set_variable = { name = bb_market_trend value = 0.45 months = 9 }
        bb_add_moral_hazard = { VALUE = 0.15 MONTHS = 18 }
    }

    # Cool the market: open-market sales and tighter discount rate.
    # Ongoing cost via bb_market_cooling_ops (9 months) — less money,
    # more authority, since tight money is the less popular course.
    option = {
        name = bb.60.b
        custom_tooltip = bb.60.b.tt
        add_modifier = {
            name = bb_market_cooling_ops
            months = 9
        }
        bb_market_jolt = { VALUE = -2 }
        set_variable = { name = bb_market_trend value = -0.45 months = 9 }
    }

    # Hold the line
    option = {
        name = bb.60.c
        default_option = yes
        custom_tooltip = bb.60.c.tt
    }
}

# ── bb.61: Market Survey ────────────────────────────────────
# Opened by the Commission a Market Survey button (which charges
# the fee). Pure information: the description is assembled live
# from the underlying trend and the current Market Health band —
# the plain-language outlook the bars only imply. No effects.

bb.61 = {
    type = country_event
    placement = ROOT
    title = bb.61.t

    # Momentum reading, then the current market band, then the
    # standing advice — three stacked paragraphs, each chosen live.
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    has_variable = bb_market_trend
                    var:bb_market_trend >= 0.5
                }
                desc = bb.61.desc.up_strong
            }
            triggered_desc = {
                trigger = {
                    has_variable = bb_market_trend
                    var:bb_market_trend >= 0.15
                }
                desc = bb.61.desc.up_mild
            }
            triggered_desc = {
                trigger = {
                    has_variable = bb_market_trend
                    var:bb_market_trend <= -0.5
                }
                desc = bb.61.desc.down_strong
            }
            triggered_desc = {
                trigger = {
                    has_variable = bb_market_trend
                    var:bb_market_trend <= -0.15
                }
                desc = bb.61.desc.down_mild
            }
            triggered_desc = {
                trigger = { always = yes }
                desc = bb.61.desc.flat
            }
        }
        # Current market band, read off the band modifier the mod
        # applies each month (rock-solid country trigger — no journal
        # scope needed). The Growing band (60-79) applies no modifier,
        # so it is the always-yes fallback.
        first_valid = {
            triggered_desc = {
                trigger = { has_modifier = bb_market_booming }
                desc = bb.61.desc.band_booming
            }
            triggered_desc = {
                trigger = { has_modifier = bb_market_stable }
                desc = bb.61.desc.band_stable
            }
            triggered_desc = {
                trigger = { has_modifier = bb_market_contracting }
                desc = bb.61.desc.band_contracting
            }
            triggered_desc = {
                trigger = { has_modifier = bb_market_crashing }
                desc = bb.61.desc.band_crashing
            }
            triggered_desc = {
                trigger = { always = yes }
                desc = bb.61.desc.band_growing
            }
        }
        desc = bb.61.desc.guidance
    }

    duration = 3

    event_image = {
        video = "unspecific_signed_contract"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    option = {
        name = bb.61.a
    }
}
