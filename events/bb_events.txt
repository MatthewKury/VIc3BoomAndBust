# ── bb.20: Adjust Fiscal Temperature — Stance Selection ─────
# Triggered by bb_fiscal_temperature_button.
# Each option sets bb_fiscal_temp (monthly Risk Stance bar delta)
# for 60 months and schedules the bb.21 expiry notice in 5 years.
# The button is greyed out while bb_fiscal_temp exists.

bb.20 = {
    type = country_event
    placement = ROOT
    title = bb.20.t
    desc  = bb.20.desc

    duration = 3

    event_image = {
        video = "unspecific_politicians_arguing"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    # Strong heating: flood the market with cheap credit
    option = {
        name = bb.20.a
        custom_tooltip = bb.20.a.tt
        set_variable = {
            name = bb_fiscal_temp
            value = 1
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Mild heating: encourage speculative lending
    option = {
        name = bb.20.b
        custom_tooltip = bb.20.b.tt
        set_variable = {
            name = bb_fiscal_temp
            value = 0.5
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Mild cooling: tighten lending standards
    option = {
        name = bb.20.c
        custom_tooltip = bb.20.c.tt
        set_variable = {
            name = bb_fiscal_temp
            value = -0.5
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Strong cooling: impose strict capital discipline
    option = {
        name = bb.20.d
        default_option = yes
        custom_tooltip = bb.20.d.tt
        set_variable = {
            name = bb_fiscal_temp
            value = -1
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }
}

# ── bb.30: Mania on the Exchange (major, market health >= 65) ─
# Rolled by bb_roll_market_event (~8%/month, 12-month cooldown).
# Speculative fever at the top of a boom: ride it or deflate it.
# Extra options unlock when Industrialists or Landowners sit in
# government. Immediate jolts use bb_market_jolt; long trends use
# the timed bb_market_trend variable read by the health bar.

bb.30 = {
    type = country_event
    placement = ROOT
    title = bb.30.t
    desc  = bb.30.desc

    duration = 3

    event_image = {
        video = "europenorthamerica_capitalists_meeting"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    # Ride the boom: surge now, harsh correction builds
    option = {
        name = bb.30.a
        custom_tooltip = bb.30.a.tt
        bb_market_jolt = { VALUE = 4 }
        set_variable = { name = bb_market_trend value = -0.6 months = 12 }
    }

    # Deflate the bubble: painful now, sounder later
    option = {
        name = bb.30.b
        default_option = yes
        custom_tooltip = bb.30.b.tt
        bb_market_jolt = { VALUE = -3 }
        set_variable = { name = bb_market_trend value = 0.4 months = 12 }
        add_radicals = {
            value = small_radicals
            pop_type = capitalists
        }
    }

    # Industrialists in government: channel the fever into industry
    option = {
        name = bb.30.c
        trigger = {
            ig:ig_industrialists = { is_in_government = yes }
        }
        custom_tooltip = bb.30.c.tt
        bb_market_jolt = { VALUE = 2 }
        set_variable = { name = bb_market_trend value = -0.2 months = 12 }
        add_loyalists = {
            value = small_radicals
            pop_type = capitalists
        }
    }

    # Landowners in government: rein in the speculating parvenus
    option = {
        name = bb.30.d
        trigger = {
            ig:ig_landowners = { is_in_government = yes }
        }
        custom_tooltip = bb.30.d.tt
        bb_market_jolt = { VALUE = -2 }
        set_variable = { name = bb_market_trend value = 0.2 months = 12 }
        add_loyalists = {
            value = small_radicals
            pop_type = aristocrats
        }
        add_radicals = {
            value = small_radicals
            pop_type = capitalists
        }
    }
}

# ── bb.11: Dissolve Monopoly — Outcome Notification ─────────
# Fires 14 days after the player makes their selection in bb.10.
# Describes the outcome; clears the outcome flag variable.

bb.11 = {
    type = country_event
    placement = ROOT
    title = bb.11.t

    duration = 3

    event_image = {
        video = "unspecific_signed_contract"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    has_variable = bb_case_strength
                    var:bb_case_strength <= -1.75
                }
                desc = bb.11.d.strong
            }
            triggered_desc = {
                trigger = {
                    has_variable = bb_case_strength
                    var:bb_case_strength >= -0.75
                }
                desc = bb.11.d.weak
            }
            triggered_desc = {
                trigger = { always = yes }
                desc = bb.11.d.base
            }
        }
    }

    option = {
        name = bb.11.a
    }
}
