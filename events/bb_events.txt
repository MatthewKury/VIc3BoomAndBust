# The button is greyed out while bb_fiscal_temp exists.

bb.20 = {
    type = country_event
    placement = ROOT
    title = bb.20.t
    desc  = bb.20.desc

    duration = 3

    event_image = {
        video = "unspecific_politicians_arguing"

    # Strong heating: flood the market with cheap credit
    option = {
        name = bb.20.a
        custom_tooltip = bb.20.a.tt
        set_variable = {
            name = bb_fiscal_temp
            value = 1
            months = 60
        }
        trigger_event = { id = bb.21 years = 5 }
    }

    # Mild heating: encourage speculative lending
    option = {
        name = bb.20.b
        custom_tooltip = bb.20.b.tt
        set_variable = {
            name = bb_fiscal_temp
            value = 0.5
            months = 60

    # Mild cooling: tighten lending standards
    option = {
        name = bb.20.c

            name = bb_fiscal_temp
            value = -0.5

# ── bb.43: Scandal at the National Bank ──────────────────────
# Rolled monthly by bb_governor_admin_pulse while the governor
# has vice traits (bb_char_gov_vice): 4%/month, 8% if a previous
# scandal was waved away (bb_scandal_lax), 6-month cooldown.
# The treasury loss and market jolt land in immediate; the
# options decide what the affair costs politically.

bb.43 = {
    type = country_event
    placement = ROOT
    title = bb.43.t
    desc  = bb.43.desc

    duration = 3

    event_image = {
        video = "europenorthamerica_capitalists_meeting"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    immediate = {
        random_scope_character = {
            limit = { has_trait = bb_bank_governor }
            save_scope_as = bb_gov
        }
        add_treasury = {
            value = gdp
            multiply = -0.004
            min = -1000000
            max = -10000
        }
        bb_market_jolt = { VALUE = -4 }
    }

    # Cover it up: the insiders now own a piece of you
    option = {
        name = bb.43.a
        custom_tooltip = bb.43.a.tt
        add_modifier = {
            name = bb_coverup_strain
            months = 24
        }
        set_variable = {
            name = bb_backroom_influence
            value = 0.5
            months = 12
        }
    }

    # Prosecute: the seat opens, the law is seen to work
    option = {
        name = bb.43.b
        default_option = yes
        custom_tooltip = bb.43.b.tt
        add_radicals = {
            value = medium_radicals
            interest_group = scope:bb_gov.interest_group
        }
        scope:bb_gov = {
            remove_trait = bb_bank_governor
        }
        bb_clear_governor_office = yes
        add_loyalists = {
            value = small_radicals
            strata = middle
        }
    }

    # Look away: cheap today, and the vices go unchecked
    option = {
        name = bb.43.c
        custom_tooltip = bb.43.c.tt
        set_variable = bb_scandal_lax
    }
}

# The jawboning option needs a voice the markets will heed.

    type = country_event
    placement = ROOT
    title = bb.50.t
    desc  = bb.50.desc

    duration = 3

    event_image = {
        video = "unspecific_signed_contract"

        custom_tooltip = bb.50.a.tt
        add_treasury = {
            value = gdp
            multiply = -0.0075
            min = -2000000
            max = -25000
        }
        bb_market_jolt = { VALUE = 6 }
        set_variable = { name = bb_market_trend value = 0.5 months = 12 }
        bb_add_moral_hazard = { VALUE = 0.4 MONTHS = 30 }

    # Targeted guarantees
    option = {
        name = bb.50.b

        set_variable = { name = bb_market_trend value = 0.3 months = 12 }

    # Jawbone the markets: costs nothing but requires a voice
